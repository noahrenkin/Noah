function renderChecklist(checkedIds) {
  const container = document.getElementById('vChecklistContainer');
  if (!container) return;
  
  const categories = {};
  CHECKLIST_ITEMS.forEach(item => {
    if (!categories[item.cat]) categories[item.cat] = [];
    categories[item.cat].push(item);
  });
  
  let total = CHECKLIST_ITEMS.length;
  let checked = checkedIds ? checkedIds.filter(id => id).length : 0;
  
  let html = `
    <div class="checklist-stats">
      <span class="stat-badge" style="color:var(--green);">✅ ${checked}/${total} points vérifiés</span>
      <span class="stat-badge" style="color:${checked/total >= 0.8 ? 'var(--green)' : checked/total >= 0.5 ? 'var(--orange)' : 'var(--red)'};">
        ${checked/total >= 0.8 ? '🟢 Bon état' : checked/total >= 0.5 ? '🟠 Moyen' : '🔴 À surveiller'}
      </span>
      <span class="stat-badge" style="color:var(--muted);">${Math.round(checked/total*100)}%</span>
    </div>
  `;
  
  html += `<div class="meter" style="margin:4px 0 12px;"><i style="width:${checked/total*100}%"></i></div>`;
  
  for (const [cat, items] of Object.entries(categories)) {
    const checkedInCat = items.filter(item => checkedIds && checkedIds.includes(item.id)).length;
    html += `
      <div class="accordion">
        <div class="accordion-header" onclick="toggleAccordion(this)">
          <span>${cat} (${checkedInCat}/${items.length})</span>
          <span style="font-size:12px;color:var(--muted);">▼</span>
        </div>
        <div class="accordion-content open">
    `;
    items.forEach(item => {
      const isChecked = checkedIds && checkedIds.includes(item.id);
      html += `
        <div class="checklist-item">
          <input type="checkbox" id="chk_${item.id}" ${isChecked ? 'checked' : ''} onchange="onChecklistChange()">
          <label for="chk_${item.id}" style="cursor:pointer;font-size:13px;">${item.label}</label>
        </div>
      `;
    });
    html += `</div></div>`;
  }
  
  container.innerHTML = html;
}

function toggleAccordion(header) {
  const content = header.nextElementSibling;
  content.classList.toggle('open');
  const arrow = header.querySelector('span:last-child');
  if (arrow) arrow.textContent = content.classList.contains('open') ? '▼' : '▶';
}

function onChecklistChange() {
  const checked = [];
  CHECKLIST_ITEMS.forEach(item => {
    const chk = document.getElementById(`chk_${item.id}`);
    if (chk && chk.checked) checked.push(item.id);
  });
  tempChecklist = checked;
  renderChecklist(checked);
}

function getChecklistState() {
  const checked = [];
  CHECKLIST_ITEMS.forEach(item => {
    const chk = document.getElementById(`chk_${item.id}`);
    if (chk && chk.checked) checked.push(item.id);
  });
  return checked;
}

// 🔍 FONCTION POUR ESTIMER LE PRIX DE MARCHÉ (inchangée)
// ============================================================
function estimerPrixMarche(brand, year, km) {
  const marqueData = prixReference[brand];
  if (!marqueData) return null;
  const annees = Object.keys(marqueData).map(Number).sort((a,b) => a-b);
  let anneeProche = annees[0], diffMin = Math.abs(Number(year) - anneeProche);
  for (let a of annees) {
    const diff = Math.abs(Number(year) - a);
    if (diff < diffMin) { diffMin = diff; anneeProche = a; }
  }
  let prixBase = marqueData[anneeProche] || 0;
  const ecartAnnee = Number(year) - anneeProche;
  if (ecartAnnee > 0) prixBase += ecartAnnee * 500;
  else if (ecartAnnee < 0) prixBase += ecartAnnee * 500;
  const kmBase = 100000, kmEcart = km - kmBase;
  const ajustementKm = Math.round(kmEcart * 0.02);
  prixBase = Math.max(prixBase + ajustementKm, 500);
  return Math.round(prixBase);
}

// ============================================================
function negotiate() {
  const ask = parseFloat($('nAsk').value) || 0;
  const brand = $('nBrand').value.trim();
  const model = $('nModel').value.trim();
  const year = parseInt($('nYear').value) || 0;
  const km = parseInt($('nKm').value) || 0;
  const rep = parseFloat($('nRep').value) || 0;
  const other = parseFloat($('nOther').value) || 0;
  const margin = parseFloat($('nMargin').value) || 1500;
  const state = $('nState').value;
  
  const prixMarche = estimerPrixMarche(brand, year, km);
  let prixBase = prixMarche || ask;
  let coeffEtat = 1;
  switch(state) {
    case 'Excellent': coeffEtat = 1.10; break;
    case 'Très bon': coeffEtat = 1.05; break;
    case 'Bon': coeffEtat = 1.00; break;
    case 'Moyen': coeffEtat = 0.90; break;
    case 'Mauvais': coeffEtat = 0.80; break;
  }
  prixBase = Math.round(prixBase * coeffEtat);
  const prixApresRep = Math.max(0, ask - (rep * 0.8) - (other * 0.9));
  const margeSecurite = Math.min(margin * 0.2, 500);
  const offreDepart = Math.max(0, Math.round(prixApresRep * 0.82) - margin + margeSecurite);
  const offreMax = Math.max(0, Math.round(prixApresRep * 0.92) - margin + (margeSecurite * 0.5));
  const coutTotal = offreDepart + rep + other;
  const beneficeDepart = prixBase - coutTotal;
  const coutTotalMax = offreMax + rep + other;
  const beneficeMax = prixBase - coutTotalMax;
  
  let arguments = [];
  if (ask > prixBase * 1.15) {
    arguments.push('💰 Le prix demandé est 15% au-dessus du marché, bonne marge de négociation.');
  } else if (ask > prixBase * 1.05) {
    arguments.push('💰 Le prix est légèrement au-dessus du marché, marge de négociation possible.');
  } else if (ask <= prixBase * 0.95) {
    arguments.push('🎯 Le prix demandé est intéressant par rapport au marché, offre à saisir !');
  }
  if (rep > 0) arguments.push('🔧 Des réparations sont nécessaires, tu peux justifier une baisse de prix.');
  if (state === 'Mauvais' || state === 'Moyen') {
    arguments.push('⚠️ L\'état général n\'est pas optimal, argument de poids pour négocier.');
  }
  if (arguments.length === 0) {
    arguments.push('💪 Le prix est correct, mais tu peux toujours tenter une négociation pour optimiser ta marge.');
  }
  
  let recommendation = '', recClass = '';
  if (offreDepart > 0 && beneficeDepart > 0) {
    recommendation = '✅ Offre réaliste avec une bonne marge. Négocie en partant de cette base.';
    recClass = 'green';
  } else if (offreDepart > 0 && beneficeDepart <= 0) {
    recommendation = '⚠️ Offre juste, vérifie les coûts et essaie d\'augmenter le prix de revente estimé.';
    recClass = 'orange';
  } else {
    recommendation = '❌ Le prix demandé est trop élevé. Passe ton chemin ou négocie très fortement.';
    recClass = 'red';
  }
  
  $('negResult').innerHTML = `
  <div class="result">
    <div style="display:grid;grid-template-columns:1fr 1fr;gap:12px;margin-bottom:14px;">
      <div>
        <div class="muted">PRIX DE MARCHÉ ESTIMÉ</div>
        <div class="scoreBig purple">${money(prixBase)}</div>
        <span class="muted" style="font-size:13px;">Basé sur ${brand} ${year} (${km.toLocaleString('fr-BE')} km)</span>
      </div>
      <div>
        <div class="muted">PRIX DEMANDÉ</div>
        <div class="scoreBig ${ask > prixBase * 1.05 ? 'red' : ask < prixBase * 0.95 ? 'green' : 'orange'}">${money(ask)}</div>
        <span class="muted" style="font-size:13px;">${ask > prixBase * 1.05 ? '⬆️ Au-dessus du marché' : ask < prixBase * 0.95 ? '⬇️ Sous le marché' : '📊 Dans la moyenne'}</span>
      </div>
    </div>
    
    <div style="border-top:1px solid var(--glassBorder);padding-top:14px;margin-top:6px;">
      <div style="display:grid;grid-template-columns:1fr 1fr;gap:12px;margin-bottom:12px;">
        <div style="background:rgba(10,4,20,.4);padding:12px;border-radius:12px;">
          <div class="muted" style="font-size:13px;">💡 OFFRE DE DÉPART</div>
          <div class="scoreBig" style="font-size:34px;color:var(--pink);">${money(offreDepart)}</div>
          <div class="muted" style="font-size:12px;">Bénéfice estimé : <span class="${beneficeDepart >= 0 ? 'green' : 'red'}">${money(beneficeDepart)}</span></div>
        </div>
        <div style="background:rgba(10,4,20,.4);padding:12px;border-radius:12px;">
          <div class="muted" style="font-size:13px;">🎯 PRIX MAXIMUM</div>
          <div class="scoreBig" style="font-size:34px;color:var(--green);">${money(offreMax)}</div>
          <div class="muted" style="font-size:12px;">Bénéfice estimé : <span class="${beneficeMax >= 0 ? 'green' : 'red'}">${money(beneficeMax)}</span></div>
        </div>
      </div>
      
      <div style="background:rgba(10,4,20,.4);padding:12px;border-radius:12px;margin-bottom:12px;">
        <div style="display:flex;justify-content:space-between;align-items:center;">
          <span class="muted">💰 BÉNÉFICE POTENTIEL</span>
          <span class="${beneficeDepart >= 0 ? 'green' : 'red'}" style="font-weight:800;font-size:18px;">${money(beneficeDepart)} → ${money(beneficeMax)}</span>
        </div>
      </div>
      
      <div style="background:rgba(10,4,20,.4);padding:12px;border-radius:12px;">
        <div class="muted" style="font-size:13px;margin-bottom:4px;">📊 DÉTAIL DES COÛTS</div>
        <div style="font-size:13px;display:grid;grid-template-columns:auto 1fr auto;gap:4px 12px;">
          <span>Prix d'achat</span><span></span><span>${money(ask)}</span>
          <span>Réparations (coeff 0.8)</span><span></span><span style="color:var(--orange);">${money(Math.round(rep * 0.8))}</span>
          <span>Autres frais</span><span></span><span style="color:var(--orange);">${money(Math.round(other * 0.9))}</span>
          <span style="border-top:1px solid var(--glassBorder);">Coût total estimé</span><span style="border-top:1px solid var(--glassBorder);"></span><span style="border-top:1px solid var(--glassBorder);font-weight:800;">${money(offreDepart + rep + other)}</span>
        </div>
      </div>
    </div>
    
    <div style="margin-top:12px;padding:12px;border-radius:12px;background:rgba(10,4,20,.3);">
      <div style="font-weight:600;margin-bottom:4px;">💬 ARGUMENTS DE NÉGOCIATION</div>
      ${arguments.map(arg => `<div style="font-size:13px;padding:3px 0;color:var(--muted);">• ${arg}</div>`).join('')}
      <div style="margin-top:6px;font-size:14px;font-weight:600;color:var(--${recClass});">${recommendation}</div>
    </div>
    
    ${!prixMarche ? '<div style="margin-top:8px;font-size:12px;color:var(--orange);">⚠️ Marque non reconnue, le prix de marché est une estimation basée sur le prix demandé.</div>' : ''}
  </div>`;
}

async function loginOrSignup(){
 const email=$('authEmail').value.trim();
 const pin=$('authPin').value.trim();
 const msg=$('authMsg1');
 if(!email||!email.includes('@')){msg.textContent='Entre un email valide.';msg.className='authMsg err';return}
 if(!/^\d{6}$/.test(pin)){msg.textContent='Le code doit faire exactement 6 chiffres.';msg.className='authMsg err';return}
 msg.textContent='Connexion...';msg.className='authMsg';
 
 // Vérification que Supabase est disponible
 if (!sb) {
   msg.textContent='Erreur : Supabase non configuré. Voir la bannière de sécurité.';
   msg.className='authMsg err';
   return;
 }
 
 let {data,error}=await sb.auth.signInWithPassword({email,password:pin});
 if(error){
   if(error.message&&error.message.toLowerCase().includes('invalid')){
     const res=await sb.auth.signUp({email,password:pin});
     if(res.error){msg.textContent='Erreur : '+res.error.message;msg.className='authMsg err';return}
     data=res.data;
   }else{
     msg.textContent='Erreur : '+error.message;msg.className='authMsg err';return
   }
 }
 currentUser=data.user;
 await afterLogin();
}

function parseAd(){

 const year=t.match(/\b(20\d{2}|19\d{2})\b/); 
 const km=t.match(/([\d\s.]{3,})\s*km/i); 
 const price=t.match(/([\d\s.]{3,})\s*€/)||t.match(/€\s*([\d\s.]{3,})/);
 if(year)$('aYear').value=year[1]; 
 if(km)$('aKm').value=km[1].replace(/[^\d]/g,''); 
 if(price)$('aPrice').value=price[1].replace(/[^\d]/g,'');
 const brands=['Audi','BMW','Mercedes','Volkswagen','Peugeot','Renault','Citroën','Toyota','Honda','Ford','Opel','Skoda','Seat','Nissan','Kia','Hyundai','Dacia','Volvo'];
 const b=brands.find(x=>new RegExp('\\b'+x+'\\b','i').test(t)); 
 if(b)$('aBrand').value=b;

}

function compressImage(file,maxDim,quality){
 return new Promise((resolve,reject)=>{  
  const img=new Image();
  const reader=new FileReader();
  reader.onload=ev=>{img.src=ev.target.result};
  reader.onerror=reject;
  img.onload=()=>{
   let w=img.width,h=img.height;
   if(w>h&&w>maxDim){h=Math.round(h*maxDim/w);w=maxDim}
   else if(h>=w&&h>maxDim){w=Math.round(w*maxDim/h);h=maxDim}
   const canvas=document.createElement('canvas');
   canvas.width=w;canvas.height=h;
   canvas.getContext('2d').drawImage(img,0,0,w,h);
   resolve(canvas.toDataURL('image/jpeg',quality));
  };
  img.onerror=reject;
  reader.readAsDataURL(file);
 });
}

function addRepair() {
  const name = document.getElementById('vRepName').value.trim();
  const price = parseFloat(document.getElementById('vRepPrice').value);
  if (!name || isNaN(price) || price <= 0) { 
    notify('Entre un nom et un prix valide (ex: Courroie, 650)'); 
    return; 
  }
  tempRepairs.push({ name, price });
  document.getElementById('vRepName').value = '';
  document.getElementById('vRepPrice').value = '';
  renderRepairs();
}

function removeRepair(index) {
  tempRepairs.splice(index, 1);
  renderRepairs();
}

function saveVehicle(id){
 const old=id?cars.find(x=>x.id===id):{};
 const repairsList = tempRepairs.map(r => ({ name: r.name, price: Number(r.price) }));
 const repairsSum = repairsList.reduce((acc, r) => acc + r.price, 0);
 const checklist = getChecklistState();

 const c={...old,id:id||Date.now(),profilePhoto:tempProfilePhoto,photo:tempProfilePhoto,obsPhotos:tempObsPhotos,brand:$('vBrand').value,model:$('vModel').value,year:+$('vYear').value||0,km:+$('vKm').value||0,buy:+$('vBuy').value||0,sell:+$('vSell').value||0,transport:+$('vTransport').value||0,admin:+$('vAdmin').value||0,clean:+$('vClean').value||0,other:+$('vOther').value||0,status:$('vStatus').value,state:$('vState').value,notes:$('vNotes').value,dateBuy:old.dateBuy||new Date().toISOString().slice(0,10),
 repairs: repairsSum,
 repairsList: repairsList,
 checklist: checklist
};
 const i=cars.findIndex(x=>x.id===c.id); 
 if(i>=0)cars[i]=c;else cars.unshift(c);
 closeModal();save();syncCarToCloud(c);
 notify('Véhicule sauvegardé 💾')
}

function totals(c){
 const cost=(+c.buy||0)+(+c.transport||0)+(+c.repairs||0)+(+c.admin||0)+(+c.clean||0)+(+c.other||0);
 const profit=(+c.sell||0)-cost;
 return {cost,profit,roi:cost?profit/cost*100:0}
}

async function checkSession(){
 if (!sb) {
   // Mode localStorage uniquement
   const savedCars = localStorage.getItem(KEY);
   if (savedCars) {
     cars = JSON.parse(savedCars);
     renderAll();
   }
   $('authScreen').style.display='grid';
   $('appRoot').style.display='none';
   return;
 }
 
 const {data}=await sb.auth.getSession();
 if(data.session){
   const {data:userCheck,error:userErr}=await sb.auth.getUser();
   if(userErr||!userCheck.user){
     await sb.auth.signOut();
     currentUser=null;
     localStorage.removeItem(KEY);
     $('authScreen').style.display='grid';
     $('appRoot').style.display='none';
     return;
   }
   currentUser=data.session.user;
   await afterLogin();
 }else{
   $('authScreen').style.display='grid';
   $('appRoot').style.display='none';
 }
}

async function loadSuggestions(){
 const el=$('suggestionsList');
 el.innerHTML='<p class="muted">Chargement...</p>';
 
 if (!sb) {
   el.innerHTML='<p class="muted">Supabase non configuré. Les suggestions ne sont pas disponibles.</p>';
   return;
 }
 
 const {data,error}=await sb.from('suggestions').select('*').order('votes',{ascending:false});
 if(error){el.innerHTML='<p class="muted">Erreur de chargement.</p>';return}
 const isAdmin=currentUser&&currentUser.email===CONFIG.ADMIN_EMAIL;
 if(!data||!data.length){el.innerHTML='<p class="muted">Aucune suggestion pour l\'instant. Sois le premier !</p>';return}
 el.innerHTML=data.map(s=>{
  const voters=s.voters||[];
  const hasVoted=currentUser&&voters.includes(currentUser.id);
  return `<div class="card" style="margin-bottom:10px;display:flex;justify-content:space-between;align-items:flex-start;gap:12px">
   <div style="flex:1">
    <p style="margin:0 0 6px">${(s.text||'').replace(/</g,'&lt;')}${s.done?' <span style="color:var(--green);font-size:12px">✓ Fait</span>':''}</p>
    <span class="muted" style="font-size:12px">par Anonyme</span>
   </div>
   <div style="display:flex;flex-direction:column;align-items:center;gap:4px">
    <button class="btn" style="padding:8px 12px" onclick="voteSuggestion(${s.id},${hasVoted})">${hasVoted?'✓':'▲'} ${s.votes||0}</button>
    ${isAdmin?`<button class="btn redBtn" style="padding:6px 10px;font-size:12px" onclick="deleteSuggestion(${s.id})">Suppr.</button>${!s.done?`<button class="btn" style="padding:6px 10px;font-size:12px" onclick="markDone(${s.id})">✓ Fait</button>`:''}`:''}
   </div>
  </div>`;
 }).join('');
}

async function postSuggestion(){
 const text=$('sugText').value.trim();
 if(!text){notify('Écris une suggestion d\'abord');return}
 if(!currentUser){notify('Connecte-toi d\'abord');return}
 
 if (!sb) {
   notify('Supabase non configuré');
   return;
 }
 
 const {error}=await sb.from('suggestions').insert({text,email:currentUser.email,user_id:currentUser.id,votes:0,voters:[]});
 if(error){notify('Erreur : '+error.message);return}
 $('sugText').value='';
 notify('Suggestion envoyée 💡');
 loadSuggestions();
}

async function voteSuggestion(id,hasVoted){
 if(!currentUser){notify('Connecte-toi d\'abord');return}
 
 if (!sb) {
   notify('Supabase non configuré');
   return;
 }
 
 const {data}=await sb.from('suggestions').select('votes,voters').eq('id',id).single();
 if(!data)return;
 let voters=data.voters||[];
 let votes=data.votes||0;
 if(hasVoted){voters=voters.filter(v=>v!==currentUser.id);votes=Math.max(0,votes-1)}
 else{voters=[...voters,currentUser.id];votes=votes+1}
 await sb.from('suggestions').update({votes,voters}).eq('id',id);
 loadSuggestions();
}

async function deleteSuggestion(id){
 if(!confirm('Supprimer cette suggestion ?'))return;
 if (!sb) return;
 await sb.from('suggestions').delete().eq('id',id);
 loadSuggestions();
}

async function markDone(id){
 if (!sb) return;
 await sb.from('suggestions').update({done:true}).eq('id',id);
 loadSuggestions();
}
